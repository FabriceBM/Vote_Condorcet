%% Overleaf entry point: the note (manuscript/), in the working-paper layout.
%%
%% Overleaf compiles from the project root, but the wrapper lives in journals/working-paper/ and loads
%% the manuscript with paths relative to that folder (../../manuscript/...). This file sets the search
%% path so those paths also resolve from the root, points the bibliography to its root-relative path,
%% then loads the wrapper unchanged (its \documentclass included). The figures are found through the
%% root-relative entries of the wrapper's \graphicspath. Nothing depends on latexmkrc, which Overleaf
%% may not apply. Same layout as Immo_S2.
%%
%% On Overleaf the main document must be this file (Menu > Main document). Any other file called main.tex
%% would compile this one instead: Overleaf writes its output to the project root, where pdfLaTeX looks
%% first for the main file. That is harmless for journals/working-paper/main.tex, which is this same
%% note; every other document needs its own name (summaries/<lang>/<kind>/<document>.tex).
%% Local builds: `latexmk main.tex` from the root, or `cd journals/working-paper && latexmk main.tex`.
%\def\FinalVersion{}
\iffalse
\documentclass{article}% never run: lets Overleaf detect this file as the main document
\fi
\makeatletter
\def\input@path{{journals/working-paper/}}
\makeatother
\newcommand{\NoteBibliography}{manuscript/references}
\input{journals/working-paper/main.tex}
